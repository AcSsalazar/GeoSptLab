Sub Grafico_1()

ActiveSheet.Unprotect Password:="up1"

Dim WsI As Worksheet
Dim CH As Chart
Dim Ser1 As Series
    '------------------------------
    Set WsI = ActiveSheet
    '********************************************************************************
    '-Funciona de excel 2007 en adelante
    'Definir el grafico
    Set CH = WsI.Shapes.AddChart( _
    XlChartType:=xlXYScatter, _
    Left:=WsI.[A56].Left, _
    Top:=WsI.[A56].Top, _
    Width:=WsI.[A56:K49].Width, _
    Height:=WsI.[A56:A71].Height).Chart
    '********************************************************************************
    'agregamos datos al grafico
   CH.SeriesCollection.NewSeries
   Set Ser1 = CH.FullSeriesCollection(1)
    With Ser1
        .Name = WsI.Range("D8")
        .Values = WsI.Range("AA93:AA122")
        .XValues = WsI.Range("Z93:Z122")
    End With
    'para poner nombre al grafico
    CH.Parent.Name = "Regresion"
    '********************************************************************************
    'Para el titulo
    CH.SetElement msoElementChartTitleAboveChart
    CH.ChartTitle.Caption = Range("D8")
    'para cambiar fuente, color y tamaño
    With CH.ChartTitle.Format.TextFrame2.TextRange.Font
        .Name = "Arial"
        .Fill.ForeColor.ObjectThemeColor = msoThemeColorDark1
        .Size = 14
    End With
    '********************************************************************************
    'PARA NOMBRES DE EJES
    'Nombre del eje categoria X
    CH.SetElement (msoElementPrimaryCategoryGridLinesMajor)
 
    CH.Axes(xlCategory, xlPrimary).HasTitle = True
    CH.Axes(xlCategory, xlPrimary).AxisTitle.Characters.Text = Range("J15")
    CH.Axes(xlCategory, xlPrimary).AxisTitle. _
    Format.TextFrame2.TextRange.Font.Fill.ForeColor.ObjectThemeColor = msoThemeColorDark1
     CH.Axes(xlCategory, xlPrimary).AxisTitle. _
    Format.TextFrame2.TextRange.Font.Name = "Arial"
    'Nombre del eje Valores Y
  
    CH.Axes(xlValue, xlPrimary).HasTitle = True
    CH.Axes(xlValue, xlPrimary).AxisTitle.Characters.Text = Range("V15")
    CH.Axes(xlValue, xlPrimary).AxisTitle. _
    Format.TextFrame2.TextRange.Font.Fill.ForeColor.ObjectThemeColor = msoThemeColorDark1
        CH.Axes(xlValue, xlPrimary).AxisTitle. _
    Format.TextFrame2.TextRange.Font.Name = "Arial"
           '********************************************************************************
      'Linea de tendencia y ecuacion
    CH.HasLegend = False
    CH.ChartColor = 13
    With CH.SeriesCollection(1).Trendlines
 .Add Type:=xlLinear, DisplayEquation:=True, DisplayRSquared:=True, Name:="Linear Trend"

 End With

ActiveSheet.Protect ("up1")

End Sub
